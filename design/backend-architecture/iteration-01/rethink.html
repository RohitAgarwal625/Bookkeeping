Revised Prisma Schema

model User {
  id              String   @id @default(cuid())
  piWalletAddress String   @unique
  piUsername       String?  @unique
  piUid           String?  @unique
  displayName     String   @default("Pioneer User")
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  contacts        Contact[]
  transactions    Transaction[]
  sessions        Session[]
  @@index([piWalletAddress])
}

model Session {
  id        String   @id @default(cuid())
  token     String   @unique
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  expiresAt DateTime
  createdAt DateTime @default(now())
  @@index([token])
}

model Contact {
  id              String   @id @default(cuid())
  name            String
  category        String
  piWalletAddress String
  userId          String
  user            User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  transactions    Transaction[]
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  @@unique([userId, piWalletAddress])
  @@index([userId])
}

model Transaction {
  id             String    @id @default(cuid())
  description    String
  amount         Decimal   @db.Decimal(18, 7)  // CHANGED from Float
  type           String
  source         String    @default("manual")
  txHash         String?
  piPaymentId    String?
  idempotencyKey String    @unique             // CHANGED: was optional
  status         String    @default("completed") // NEW
  expiresAt      DateTime?                      // NEW: TTL
  deletedAt      DateTime?                      // NEW: soft delete
  timestamp      DateTime  @default(now())
  userId         String
  user           User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  contactId      String?
  contact        Contact?  @relation(fields: [contactId], references: [id], onDelete: SetNull) // CHANGED
  @@index([userId])
  @@index([userId, contactId])
  @@index([userId, timestamp])
  @@index([idempotencyKey])
  @@index([status, expiresAt])  // NEW
}

Revised Auth Flow (Phase 1-3)

  1. User enters wallet address
  2. Frontend: GET /api/users/:walletAddress
  3. Backend: upsert user + create Session (UUID token, 24h TTL)
  4. Returns { user, sessionToken }
  5. Frontend stores sessionToken in localStorage
  6. All requests: Authorization: Bearer {sessionToken}
  7. Middleware: validate token → attach req.user
  8. Phase 4: replace with Pi OAuth JWT

Remaining Uncertainties

  • Pi wallet address format — low impact, permissive validation for now
  • Pi Platform API reconciliation behavior — Phase 4 sandbox testing
  • Railway pricing post-trial — $5/mo expected, Neon as fallback

None blocking Phase 1-3 implementation.
