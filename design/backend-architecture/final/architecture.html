Database Schema (Prisma)

// backend/prisma/schema.prisma

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model User {
  id              String        @id @default(cuid())
  piWalletAddress String        @unique
  piUsername       String?       @unique
  piUid           String?       @unique
  displayName     String        @default("Pioneer User")
  createdAt       DateTime      @default(now())
  updatedAt       DateTime      @updatedAt
  contacts        Contact[]
  transactions    Transaction[]
  sessions        Session[]

  @@index([piWalletAddress])
}

model Session {
  id        String   @id @default(cuid())
  token     String   @unique
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  expiresAt DateTime
  createdAt DateTime @default(now())

  @@index([token])
  @@index([userId])
}

model Contact {
  id              String        @id @default(cuid())
  name            String
  category        String        // "individual" | "business"
  piWalletAddress String
  userId          String
  user            User          @relation(fields: [userId], references: [id], onDelete: Cascade)
  transactions    Transaction[]
  createdAt       DateTime      @default(now())
  updatedAt       DateTime      @updatedAt

  @@unique([userId, piWalletAddress])
  @@index([userId])
}

model Transaction {
  id             String    @id @default(cuid())
  description    String
  amount         Decimal   @db.Decimal(18, 7)
  type           String    // "credit" | "debit"
  source         String    @default("manual")   // "manual" | "pi_network"
  txHash         String?
  piPaymentId    String?
  idempotencyKey String    @unique
  status         String    @default("completed") // pending|completed|failed|cancelled|expired
  expiresAt      DateTime?
  deletedAt      DateTime?
  timestamp      DateTime  @default(now())
  userId         String
  user           User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  contactId      String?
  contact        Contact?  @relation(fields: [contactId], references: [id], onDelete: SetNull)

  @@index([userId])
  @@index([userId, contactId])
  @@index([userId, timestamp])
  @@index([idempotencyKey])
  @@index([status, expiresAt])
}

Auth Middleware

// backend/src/middleware/auth.ts
export async function sessionAuth(req, res, next) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  if (!token) return res.status(401).json({ error: 'UNAUTHORIZED' });

  const session = await prisma.session.findUnique({
    where: { token },
    include: { user: true }
  });

  if (!session || session.expiresAt < new Date()) {
    return res.status(401).json({ error: 'SESSION_EXPIRED' });
  }

  req.user = session.user;
  next();
}

Zod Schema Example (Transaction)

import { z } from 'zod';
import xss from 'xss';

const sanitize = (val: string) => xss(val, { whiteList: {}, stripIgnoreTag: true });

export const createTransactionSchema = z.object({
  body: z.object({
    description: z.string().min(1).max(500).transform(sanitize),
    amount: z.number().positive().min(0.0000001).max(1_000_000),
    type: z.enum(['credit', 'debit']),
    contactId: z.string().optional(),
    timestamp: z.string().datetime().optional(),
    idempotencyKey: z.string().uuid(),  // REQUIRED
  })

Security Layers

  1. CORS: Whitelist localhost:5173 (dev) + bookkeeping.vercel.app (prod)
  2. Rate Limiting: IP: 200/min (unauthenticated), User: 100/min (read), 20/min (write)
  3. Input Sanitization: xss package strips HTML from all text fields via Zod transform
  4. SQL Injection: Prevented by Prisma parameterized queries
  5. User Isolation: Every query includes WHERE userId = req.user.id
  6. Ownership Verification: ContactId validated against req.user.id before use
  7. Helmet.js: X-Content-Type-Options, X-Frame-Options, etc.

Data Flow

Contact List (with computed totals)

// Single query solves G-010 and G-025 (N+1)
const contacts = await prisma.$queryRaw`
  SELECT c.id, c.name, c.category, c."piWalletAddress",
    c."createdAt", c."updatedAt",
    COALESCE(SUM(CASE WHEN t.type='credit' THEN t.amount ELSE 0 END), 0)::float AS "totalCredit",
    COALESCE(SUM(CASE WHEN t.type='debit' THEN t.amount ELSE 0 END), 0)::float AS "totalDebit",
    MAX(t.timestamp) AS "lastSeen"
  FROM "Contact" c
  LEFT JOIN "Transaction" t ON t."contactId" = c.id
    AND t."userId" = ${userId}
    AND t."deletedAt" IS NULL
  WHERE c."userId" = ${userId}
  GROUP BY c.id
  ORDER BY "lastSeen" DESC NULLS LAST
`;

Balance Summary

const summary = await prisma.$queryRaw`
  SELECT
    COALESCE(SUM(CASE WHEN type='credit' THEN amount ELSE 0 END), 0)::float AS "totalCredit",
    COALESCE(SUM(CASE WHEN type='debit' THEN amount ELSE 0 END), 0)::float AS "totalDebit",
    COUNT(*)::int AS "transactionCount"
  FROM "Transaction"
  WHERE "userId" = ${userId}
    AND "deletedAt" IS NULL
    AND status = 'completed'
`;

Reconciliation Job (Phase 4)
